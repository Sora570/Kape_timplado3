/* Low stock and out-of-stock handling */
function renderLowStockPanel(){
  const low = products.filter(p => Number(p.qty) < LOW_THRESHOLD && Number(p.qty) > 0);
  const out = products.filter(p => Number(p.qty) === 0);
  const listEl = $('lowList');
  if(listEl == null) return;
  if(low.length === 0 && out.length === 0){ listEl.innerHTML = '<div class="tiny muted">No low stock</div>'; $('lowBadge').style.display = 'none'; return; }

  let html = '';
  if(low.length){
    html += low.map(p => `<div class="notif-item"><strong>${escapeHtml(p.name)}</strong> • ${p.qty} left</div>`).join('');
  }
  if(out.length){
    html += out.map(p => `<div class="notif-item" style="color:${p.qty===0? 'var(--danger)':''}"><strong>${escapeHtml(p.name)}</strong> • Out of stock</div>`).join('');
  }
  listEl.innerHTML = html;

  // push notifications for low and out-of-stock if not already present
  low.forEach(p => {
    const exists = notifications.some(n => n.type === 'lowStock' && n.text.includes(p.name));
    if(!exists) pushNotification('lowStock', `Low stock: ${p.name} (${p.qty})`);
  });
  out.forEach(p => {
    const exists = notifications.some(n => n.type === 'outStock' && n.text.includes(p.name));
    if(!exists) pushNotification('outStock', `Out of stock: ${p.name}`);
  });

  const totalAlerts = low.length + out.length;
  if(totalAlerts > 0){ $('lowBadge').textContent = totalAlerts; $('lowBadge').style.display = ''; } else { $('lowBadge').style.display = 'none'; }
}

function renderProducts(){
  const grid = $('productGrid'); if(!grid) return; grid.innerHTML = '';
  const large = $('productsLarge'); if(large) large.innerHTML = '';
  const q = $('searchInput') ? $('searchInput').value.trim().toLowerCase() : '';

  products.forEach((p, idx) => {
    if(filter && p.category !== filter) return;
    if(q && !p.name.toLowerCase().includes(q) && !(p.addons || '').toLowerCase().includes(q)) return;

    // small card
    const card = document.createElement('div');
    card.className = 'product-card';
    card.innerHTML = `
      <div class="header">
        <div>
          <div class="name">${escapeHtml(p.name)}</div>
          <div class="meta">${escapeHtml(p.category)} ${p.size ? '• ' + escapeHtml(p.size) : ''}</div>
        </div>
        <div style="text-align:right">
          <div style="font-weight:800;color:var(--primary)">${formatMoney(p.price)}</div>
          <div class="meta">Stock: ${p.qty}</div>
        </div>
      </div>
      <div class="tiny muted">${p.addons ? (p.category==='food' ? 'Flavors: ' : 'Add-ons: ') + escapeHtml(p.addons) : ''}</div>
      <div class="actions">
        <button class="btn" onclick="handleAddClick(${idx})">Add</button>
        <button class="btn secondary" onclick="editProduct(${idx})">Edit</button>
      </div>
    `;
    grid.appendChild(card);

    // large center listing
    if(large){
      const lg = card.cloneNode(true);
      lg.querySelector('.btn').setAttribute('onclick', `handleAddClick(${idx})`);
      lg.querySelector('.btn.secondary').setAttribute('onclick', `editProduct(${idx})`);
      large.appendChild(lg);
    }
  });

  // If no products, show helpful note
  if(grid && grid.children.length === 0){
    grid.innerHTML = '<div class="tiny muted">No products to show. Click "+ Add" to create a product.</div>';
  }
}

/* Open addon modal for product with addons/flavors */
function handleAddClick(index){
  const p = products[index];
  if(!p) return;
  if(Number(p.qty) <= 0){ alert('Out of stock'); return; }
  const arr = (p.addons && p.addons.trim()) ? p.addons.split(',').map(s=>s.trim()).filter(Boolean) : [];
  if(arr.length === 0){
    addToCart(index, []);
    return;
  }
  pendingAddonIndex = index;
  pendingSelectedAddons = [];
  $('addonsTitle').textContent = `Choose ${p.category === 'food' ? 'flavors' : 'add-ons'} for "${p.name}"`;
  const opts = $('addonsOptions'); opts.innerHTML = '';
  arr.forEach((a,i) => {
    const id = `opt_${index}_${i}`;
    const div = document.createElement('div');
    div.style.marginBottom = '8px';
    div.innerHTML = `<label style="display:flex;align-items:center;gap:8px"><input type="checkbox" id="${id}" value="${escapeHtml(a)}" onchange="togglePending(this)"> ${escapeHtml(a)}</label>`;
    opts.appendChild(div);
  });
  $('addonsModal').style.display = 'flex';
}
function togglePending(chk){ const v = chk.value; if(chk.checked){ if(!pendingSelectedAddons.includes(v)) pendingSelectedAddons.push(v);} else pendingSelectedAddons = pendingSelectedAddons.filter(x=>x!==v); }
function confirmAddons(){ if(pendingAddonIndex === null) return closeAddons(); addToCart(pendingAddonIndex, pendingSelectedAddons.slice()); closeAddons(); }
function closeAddons(){ pendingAddonIndex = null; pendingSelectedAddons = []; $('addonsModal').style.display = 'none'; }

/* Add to cart logic with combining same product + addons */
function addToCart(prodIndex, addons){
  const p = products[prodIndex];
  if(!p) return;
  const key = cart.find(c => c.name === p.name && (c.size||'') === (p.size||'') && (c.addons||[]).join('|') === (addons||[]).join('|'));
  if(key){ key.qty += 1; } else {
    cart.push({name:p.name, price:Number(p.price), qty:1, size:p.size||'', addons:addons||[], category:p.category});
  }
  renderCart();
}

function renderCart(){
  const box = $('cartItems'); if(!box) return; box.innerHTML = '';
  if(cart.length === 0){ box.innerHTML = '<div class="tiny muted">Cart is empty.</div>'; $('cartCount').textContent = '0 items'; $('cartTotal').textContent = formatMoney(0); return; }
  let total = 0, count = 0;
  cart.forEach((c, idx) => {
    total += c.price * c.qty; count += c.qty;
    const row = document.createElement('div');
    row.className = 'cart-row';
    row.innerHTML = `<div>
      <div style="font-weight:700">${escapeHtml(c.name)}${c.size?` • ${escapeHtml(c.size)}`:''}</div>
      <div class="tiny muted">${(c.addons && c.addons.length) ? 'Add-ons: ' + escapeHtml(c.addons.join(', ')) : ''}</div>
      <div class="tiny muted">₱${Number(c.price).toFixed(2)} × ${c.qty}</div>
    </div>
    <div style="display:flex;flex-direction:column;gap:6px;align-items:center">
      <div style="display:flex;flex-direction:column;gap:6px">
        <button class="btn secondary" onclick="changeQty(${idx}, -1)">-</button>
        <button class="btn secondary" onclick="changeQty(${idx}, 1)">+</button>
      </div>
      <button class="btn" style="background:var(--danger);padding:6px;border-radius:8px;margin-top:6px" onclick="removeCart(${idx})">Remove</button>
    </div>`;
    box.appendChild(row);
  });
  $('cartCount').textContent = `${count} items`;
  $('cartTotal').textContent = formatMoney(total);
}
function changeQty(idx, delta){ if(!cart[idx]) return; cart[idx].qty = Math.max(1, cart[idx].qty + delta); renderCart(); }
function removeCart(idx){ cart.splice(idx,1); renderCart(); }
function clearCart(){ if(confirm('Clear cart?')){ cart = []; renderCart(); } }

function openCheckout(){
  if(cart.length === 0){ alert('Cart is empty'); return; }
  const total = cart.reduce((s,c) => s + c.price * c.qty, 0);
  $('checkoutTotal').textContent = formatMoney(total);
  $('custName').value = '';
  $('custNotes').value = '';
  $('checkoutModal').style.display = 'flex';
}
function closeCheckout(){ $('checkoutModal').style.display = 'none'; }
function confirmCheckout(){
  const mode = $('payMode').value;
  const customer = $('custName').value.trim();
  const notes = $('custNotes').value.trim();
  const id = Date.now();
  const total = cart.reduce((s,c) => s + c.price * c.qty, 0);
  const order = {id, date: nowISO(), total, payment: mode, customer, notes, items: cart.map(i=>({...i})), completed: true};
  orders.push(order);

  // decrement stock quantities
  order.items.forEach(it => {
    const p = products.find(x => x.name === it.name && (x.size||'') === (it.size||''));
    if(p) p.qty = Math.max(0, Number(p.qty) - Number(it.qty));
  });

  pushNotification('newOrder', `New order #${id} • ${formatMoney(total)} • ${mode}`);
  saveAll();
  cart = [];
  closeCheckout();
  renderAll();

  // Immediately open receipt modal for printed copy (optional)
  openReceiptModal(order);
}

function toggleProdFields(){
  const cat = $('prodCategory').value;
  // Show size row for beverage or non-coffee
  const isDrink = (cat === 'beverage' || cat === 'noncoffee');
  $('prodSizeRow').style.display = isDrink ? '' : 'none';
  $('prodAddonsBeverage').style.display = cat === 'beverage' ? '' : 'none';
  $('prodAddonsFood').style.display = cat === 'food' ? '' : 'none';
}
function saveProduct(){
  const name = $('prodName').value.trim();
  const category = $('prodCategory').value;
  const size = (category === 'beverage' || category === 'noncoffee') ? $('prodSize').value : '';
  const addons = category === 'beverage' ? $('prodAddonsBeverageInput').value.trim() : $('prodAddonsFoodInput').value.trim();
  const price = parseFloat($('prodPrice').value) || 0;
  const qty = parseInt($('prodQty').value) || 0;
  if(!name){ alert('Please enter product name'); return; }
  const prod = {name, category, size, addons, price, qty};
  if(editingProductIndex === null){ products.push(prod); } else { products[editingProductIndex] = prod; }
  saveAll(); closeProductModal(); renderAll();
}

/* Edit / delete / duplicate / restock helpers */
function editProduct(idx){
  const p = products[idx]; if(!p) return;
  editingProductIndex = idx;
  $('productModalTitle').textContent = 'Edit Product';
  $('prodName').value = p.name;
  $('prodCategory').value = p.category;
  $('prodSize').value = p.size || '22 oz';
  $('prodAddonsBeverageInput').value = p.addons || '';
  $('prodAddonsFoodInput').value = p.addons || '';
  $('prodPrice').value = p.price;
  $('prodQty').value = p.qty;
  toggleProdFields();
  $('productModal').style.display = 'flex';
}
function deleteProduct(idx){
  if(confirm('Delete this product?')){ products.splice(idx,1); saveAll(); renderAll(); }
}
function duplicate(idx){
  const p = products[idx]; if(!p) return;
  products.push({...p, name: p.name + ' (copy)'});
  saveAll(); renderAll();
}
function restock(idx){
  const qty = prompt('Enter quantity to add:', '10');
  const n = parseInt(qty);
  if(!isNaN(n) && n > 0){ products[idx].qty = Number(products[idx].qty || 0) + n; saveAll(); renderAll(); }
}

function isSameLocalDay(isoDateStr){
  try {
    const d = new Date(isoDateStr);
    const now = new Date();
    return d.getFullYear() === now.getFullYear() && d.getMonth() === now.getMonth() && d.getDate() === now.getDate();
  } catch(e){ return false; }
}
function renderDaily(){

  const today = new Date();
  const dateLabel = `${today.getFullYear()}-${('0'+(today.getMonth()+1)).slice(-2)}-${('0'+today.getDate()).slice(-2)}`;
  label.textContent = 'Date: ' + dateLabel;

  const dailyOrders = orders.filter(o => isSameLocalDay(o.date));
  const body = $('dailyBody'); if(!body) return; body.innerHTML = '';

/* Generate PDF for daily sales with summary + full order list */
function generateDailySalesPDF(){
  if(!currentUser || currentUser.role !== 'admin'){ alert('Only admins can download daily sales'); return; }
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF();
  const now = new Date();
  const dateLabel = `${now.getFullYear()}-${('0'+(now.getMonth()+1)).slice(-2)}-${('0'+now.getDate()).slice(-2)}`;
  const dailyOrders = orders.filter(o => isSameLocalDay(o.date));
  const totalSales = dailyOrders.reduce((s,o) => s + Number(o.total), 0);
  const totalOrders = dailyOrders.length;
  const tally = {};
  dailyOrders.forEach(o => o.items.forEach(it => { tally[it.name] = (tally[it.name]||0) + Number(it.qty); }));
  const top = Object.keys(tally).sort((a,b)=>tally[b]-tally[a])[0] || '—';

  doc.setFont("helvetica", "bold");
  doc.setFontSize(16);
  doc.text("Kape Timplado — Daily Sales Record", 14, 18);
  doc.setFont("helvetica", "normal");
  doc.setFontSize(10);
  doc.text(`Date: ${dateLabel}`, 14, 26);
  doc.text(`Generated: ${now.toLocaleString()}`, 14, 32);
  doc.line(14, 34, 196, 34);

  doc.setFont("helvetica", "bold");
  doc.text("Summary", 14, 42);
  doc.setFont("helvetica", "normal");
  doc.text(`Total Sales: ₱${totalSales.toFixed(2)}`, 14, 48);
  doc.text(`Total Orders: ${totalOrders}`, 14, 54);
  doc.text(`Top Product: ${top}`, 14, 60);

  doc.setFont("helvetica", "bold");
  doc.text("Orders (detailed)", 14, 72);
  doc.setFont("helvetica", "normal");

  let y = 78;
  if(dailyOrders.length === 0){
    doc.text("No transactions for today", 14, y);
  } else {
    dailyOrders.slice().reverse().forEach(o => {
      if(y > 270){ doc.addPage(); y = 20; }
      doc.setFont("helvetica", "bold");
      doc.text(`#${o.id} • ${new Date(o.date).toLocaleTimeString()}`, 14, y);
      y += 6;
      doc.setFont("helvetica", "normal");
      const itemsText = o.items.map(it => `${it.name} x${it.qty}`).join(', ');
      const splitItems = doc.splitTextToSize(itemsText, 160);
      doc.text(splitItems, 16, y);
      y += splitItems.length * 6;
      doc.text(`Total: ₱${Number(o.total).toFixed(2)} • Payment: ${o.payment}`, 16, y);
      y += 10;
    });
  }

  doc.save(`Daily_Sales_${dateLabel}.pdf`);
}

function switchTabFromNav(ev){
  if(getComputedStyle(ev.currentTarget).display === 'none') return; // hidden nav item
  document.querySelectorAll('.nav-item').forEach(i=>i.classList.remove('active'));
  ev.currentTarget.classList.add('active');
  const id = ev.currentTarget.getAttribute('data-tab');

  // Hide all tab panels
  document.querySelectorAll('[role="tabpanel"]').forEach(t => t.style.display = 'none');

  // Show the target panel
  const panel = document.getElementById(id);
  if(panel) panel.style.display = '';

  // PRODUCTS: show fade-in inner panels; otherwise hide them (fade-out)
  if(id === 'tabProducts'){
    showProductsPanel();
  } else {
    hideProductsPanel();
  }

  // Render content of specific tabs when activated
  if(id === 'tabOrders') renderOrders();
  if(id === 'tabInventory') renderInventory();
  if(id === 'tabSettings') renderSettings();
  if(id === 'tabCloseout') showCloseout();
  if(id === 'tabDashboard') renderDashboard();
  if(id === 'tabHistory') renderHistory();
  if(id === 'tabDaily') renderDaily();

  closeMobileNav();
}

/* -------------------------
   Dashboard rendering & jsPDF PDF
   ------------------------- */
function renderDashboard(){
  if(!currentUser || currentUser.role !== 'admin'){
    $('dashboardAlert').textContent = 'Access restricted — Admins only.';
    return;
  }
  $('dashboardAlert').textContent = '';
  const totalSales = orders.reduce((s,o) => s + Number(o.total), 0);
  $('dashTotalSales').textContent = formatMoney(totalSales);
  $('dashTotalOrders').textContent = orders.length;
  const lowCount = products.filter(p => Number(p.qty) < LOW_THRESHOLD).length;
  $('dashLowCount').textContent = lowCount;
  // Top product by total sold (based on orders)
  const tally = {};
  orders.forEach(o => o.items.forEach(it => { tally[it.name] = (tally[it.name]||0) + Number(it.qty); }));
  const top = Object.keys(tally).sort((a,b) => tally[b] - tally[a])[0] || '—';
  $('dashTopProduct').textContent = top;

  // Stock summary list
  const stockSum = $('stockSummary');
  if(stockSum) stockSum.innerHTML = products.map(p => `<div style="display:flex;justify-content:space-between;margin-bottom:6px"><div>${escapeHtml(p.name)}</div><div>${p.qty}</div></div>`).join('');

  // Simple sales chart: last up to 10 orders, plot totals (canvas)
  const canvas = document.getElementById('salesChart');
  if(!canvas) return;
  const ctx = canvas.getContext('2d');
  ctx.clearRect(0,0,ctx.canvas.width,ctx.canvas.height);
  const recent = orders.slice(-10);
  if(recent.length === 0){
    ctx.fillStyle = '#777';
    ctx.font = '14px sans-serif';
    ctx.fillText('No sales data yet', 20, 40);
    return;
  }
  const labels = recent.map(o => new Date(o.date).toLocaleTimeString());
  const data = recent.map(o => Number(o.total));
  const padding = 20;
  const cw = ctx.canvas.width;
  const ch = ctx.canvas.height;
  const chartW = cw - padding*2;
  const chartH = ch - padding*2;
  const max = Math.max(...data) || 1;
  const barW = chartW / data.length * 0.7;
  const gap = (chartW / data.length) * 0.3;
  ctx.strokeStyle = 'rgba(0,0,0,0.06)';
  ctx.lineWidth = 1;
  ctx.beginPath();
  for(let i=0;i<=4;i++){
    const y = padding + (chartH / 4) * i;
    ctx.moveTo(padding, y);
    ctx.lineTo(padding + chartW, y);
  }
  ctx.stroke();
  data.forEach((v,i) => {
    const x = padding + i*(barW + gap) + gap/2;
    const h = (v / max) * (chartH - 10);
    const y = padding + chartH - h;
    ctx.fillStyle = '#6d4b3e';
    ctx.fillRect(x, y, barW, h);
    ctx.fillStyle = '#444';
    ctx.font = '11px sans-serif';
    ctx.fillText('₱' + v.toFixed(0), x, y - 6);
    ctx.save();
    ctx.translate(x + barW/2, padding + chartH + 14);
    ctx.rotate(-0.6);
    ctx.fillText(labels[i], 0, 0);
    ctx.restore();
  });
}

/* Generate PDF for transactions history (all orders) */
function generateTransactionsPDF(){
  if(!currentUser || currentUser.role !== 'admin'){ alert('Only admins can download transactions'); return; }
  const { jsPDF } = window.jspdf;
  const doc = new jsPDF();
  const date = new Date().toLocaleString();

  doc.setFont("helvetica", "bold"); doc.setFontSize(16);
  doc.text("Kape Timplado — Transactions", 14, 18);
  doc.setFontSize(10); doc.setFont("helvetica", "normal");
  doc.text(`Generated: ${date}`, 14, 24);
  doc.line(14, 26, 196, 26);

  let y = 34;
  doc.setFont("helvetica", "bold");
  doc.text("Order ID", 14, y);
  doc.text("Date", 40, y);
  doc.text("Items", 84, y);
  doc.text("Total", 150, y);
  doc.text("Payment", 172, y);
  y += 6;
  doc.setFont("helvetica", "normal");

  orders.slice().reverse().forEach(o => {
    if(y > 270){ doc.addPage(); y = 20; }
    const itemsText = o.items.map(it => `${it.name} x${it.qty}`).join(', ');
    doc.text(String(o.id), 14, y);
    doc.text(new Date(o.date).toLocaleString(), 40, y);
    const splitItems = doc.splitTextToSize(itemsText, 60);
    doc.text(splitItems, 84, y);
    doc.text(`₱${Number(o.total).toFixed(2)}`, 150, y);
    doc.text(o.payment, 172, y);
    const lines = splitItems.length;
    y += lines * 6;
    y += 4;
  });

  doc.save("Kape_Timplado_Transactions.pdf");
}

(function init(){
  // Apply saved sidebar width/state
  const storedWidth = parseInt(localStorage.getItem('kt_sidebar_width') || 320);
  setProductsSidebarWidth(storedWidth);
  applySavedSidebarState();

  // Start with products inner hidden (so layout unchanged)
  hideProductsPanel();

  renderAll();
  // Hide modals initially
  document.querySelectorAll('.modal-backdrop').forEach(m => m.style.display = 'none');
  // Clicking outside panels hides them
  document.addEventListener('click', function(e){
    if(!e.target.closest('#notifBtn') && !e.target.closest('#notifPanel')){ if($('notifPanel')) $('notifPanel').style.display = 'none'; }
    if(!e.target.closest('#lowBtn') && !e.target.closest('#lowPanel')){ if($('lowPanel')) $('lowPanel').style.display = 'none'; }
  });
  // Default nav
  activateNav('tabDashboard');

  // Responsive nav toggle visibility
  const mt = document.getElementById('mobileNavToggle');
  function responsiveToggle(){ if(window.innerWidth <= 1200){ mt.style.display = 'inline-block'; } else { mt.style.display = 'none'; closeMobileNav(); } }
  responsiveToggle();
  window.addEventListener('resize', responsiveToggle);

  // double click toggle to set wide 420 or 320 for sidebar
  const tbtn = $('toggleSidebarBtn');
  if(tbtn){
    tbtn.ondblclick = function(){
      const current = parseInt(getComputedStyle(document.documentElement).getPropertyValue('--sidebar-width'));
      const neww = current < 400 ? 420 : 320;
      setProductsSidebarWidth(neww);
      document.documentElement.style.setProperty('--sidebar-width', neww + 'px');
      $('productsSidebar').classList.remove('collapsed');
      localStorage.setItem('kt_sidebar_expanded','true');
      $('sidebarStateHint').textContent = '(expanded)';
    };
  }
})();
</script>
